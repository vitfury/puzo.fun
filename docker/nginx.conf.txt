server {
    listen 80;
    listen [::]:80;
    server_name melody.ninja www.melody.ninja localhost;

    # Redirect HTTP to HTTPS in production
    # return 301 https://$server_name$request_uri;

    # Root directory for Laravel public folder
    root /var/www/html/public;
    index index.php index.html;

    # Increase client body size for image uploads
    client_max_body_size 20M;

    # API routes (Laravel backend)
    location /api {
        try_files $uri $uri/ /index.php?$query_string;
    }

    # Auth routes (Laravel backend)
    location /auth {
        try_files $uri $uri/ /index.php?$query_string;
    }

    # Storage routes (Laravel backend)
    location /storage {
        try_files $uri $uri/ /index.php?$query_string;
    }

    # PHP-FPM configuration
    location ~ \.php$ {
        try_files $uri =404;
        fastcgi_split_path_info ^(.+\.php)(/.+)$;
        fastcgi_pass app:9000;
        fastcgi_index index.php;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_param PATH_INFO $fastcgi_path_info;
        fastcgi_read_timeout 300;
    }

    # React frontend (proxy to port 3000 in development)
    location / {
        proxy_pass http://frontend:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Deny access to hidden files
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }

    # Gzip compression
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/x-javascript application/xml+rss application/json application/javascript;
}
